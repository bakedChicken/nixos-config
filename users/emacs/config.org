* Eglot
** Nix

Every code buffer starts Eglot

  (defun artur/eglot-ensure ()
    (unless (derived-mode-p 'lisp-data-mode)
      (eglot-ensure)))

  (use-package eglot
    :hook
    (eglot-managed-mode . eglot-inlay-hints-mode)
    (before-save . eglot-format)
    (prog-mode . artur/eglot-ensure))

** Org source blocks

#+begin_src emacs-lisp
  (defun artur/org-src-eglot ()
    (let ((org-file (buffer-file-name (marker-buffer org-src--beg-marker)))
          (tangle (alist-get :tangle (nth 2 org-src--babel-info))))
      (when (and org-file org-src--babel-info (derived-mode-p 'prog-mode) (not (derived-mode-p 'lisp-data-mode)))
        (setq buffer-file-name
              (if (member tangle '(nil "no" "yes"))
                  (concat (file-name-sans-extension org-file) ".org-src." (car org-src--babel-info))
                (expand-file-name tangle (file-name-directory org-file))))
        (eglot-ensure))))

  (add-hook 'org-src-mode-hook #'artur/org-src-eglot)
#+end_src
