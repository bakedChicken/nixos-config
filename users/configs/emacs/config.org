* UI configuration
** Turn off autosave files

#+begin_src emacs-lisp
  (setq auto-save-default nil)
  (setq make-backup-files nil)
#+end_src

** Turn off headbars

#+begin_src emacs-lisp
  (menu-bar-mode -1)
  (scroll-bar-mode -1)
  (tool-bar-mode -1)
#+end_src

** Fonts

#+begin_src emacs-lisp
  (let ((mono-spaced-font "FiraCode Nerd Font")
      (proportionately-spaced-font "Sans"))
  (set-face-attribute 'default nil :family mono-spaced-font :height 125)
  (set-face-attribute 'fixed-pitch nil :family mono-spaced-font :height 1.0)
  (set-face-attribute 'variable-pitch nil :family proportionately-spaced-font :height 1.0))
#+end_src

** Display Line Numbers

#+begin_src emacs-lisp
  (global-display-line-numbers-mode 1)
  (global-visual-line-mode t)
  (setq next-line-add-newlines t) ; Add new line in the end of each buffer during save
  (column-number-mode 1)
#+end_src

** Enable some mods by default

#+begin_src emacs-lisp
  (delete-selection-mode 1)    ; Replaces selected text on paste
  (global-auto-revert-mode 1)  ; Automatically reloads files from disk
  (show-paren-mode 1)          ; Highlights parentheses
#+end_src

** Indentation

#+begin_src emacs-lisp
  (setq standard-indent 2)
  (setq indent-tabs-mode nil)
#+end_src

** Dired

#+begin_src emacs-lisp
  (use-package dired
    :ensure nil
    :commands (dired)

      (setq ls-lisp-dirs-first t)
      (setq ls-lisp-use-insert-directory-program nil))

Fuzzy search commands in the minibuffer

#+begin_src emacs-lisp

    :config

#+end_src

** Corfu

Popup autocompletion for the buffer

#+begin_src emacs-lisp
  (use-package corfu
    :hook (after-init . global-corfu-mode)
    :bind (:map corfu-map ("<tab>" . corfu-complete))
    :config
    (setq corfu-cycle t)
    (corfu-popupinfo-mode 1))
#+end_src
