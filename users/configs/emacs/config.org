* UI configuration
** Turn off autosave files

#+begin_src emacs-lisp
  (setq auto-save-default nil)
  (setq make-backup-files nil)
#+end_src

** Display Line Numbers and Enable some mods by default

#+begin_src emacs-lisp
  (global-display-line-numbers-mode 1)
  (global-visual-line-mode t)
  (column-number-mode 1)
  (delete-selection-mode 1)    ; Replaces selected text on paste
  (global-auto-revert-mode 1)  ; Automatically reloads files from disk
  (show-paren-mode 1)          ; Highlights parentheses
#+end_src

** Dired

#+begin_src emacs-lisp
  (use-package dired
    :ensure nil
    :commands (dired)
    :hook (dired-mode . hl-line-mode) ; enable current-line highlighting
    :config
      (setq delete-by-moving-to-trash t)
      (setq ls-lisp-dirs-first t)
      (setq ls-lisp-use-string-collate nil)
      (setq ls-lisp-UCA-like-collation nil)
      (setq ls-lisp-use-insert-directory-program nil))
#+end_src

** Orderless

Fuzzy search commands in the minibuffer

#+begin_src emacs-lisp
  (use-package orderless
    :config
    (setq completion-styles '(orderless basic))
    (setq completion-category-defaults nil)
    (setq completion-category-overrides nil))
#+end_src

** Corfu

Popup autocompletion for the minibuffer

#+begin_src emacs-lisp
  (use-package corfu
    :hook (after-init . global-corfu-mode)
    :bind (:map corfu-map ("<tab>" . corfu-complete))
    :config
    (setq corfu-cycle t)
    (corfu-popupinfo-mode 1))
#+end_src
